const plural = (n: number, one: string, many = `${one}s`) => `${fmt(n)} ${n === 1 ? one : many}`;
const share = (n: number, of: number) => (of > 0 ? `${nf1.format((n / of) * 100)}%` : "—");
const eur2 = (n: number) => `€${n.toFixed(2)}`;

/** Leading words + pack size identify a product line ("Karl Fazer … 180g"); same brand and comparison group. */
const sizeOf = (name: string) => name.match(/(\d+(?:[.,]\d+)?\s?(?:g|kg|ml|l|dl|cl))\b/i)?.[1]?.replace(/\s/g, "").toLowerCase() ?? null;
function siblingGroups(products: AccountProduct[]): AccountProduct[][] {
  const groups = new Map<string, AccountProduct[]>();
  for (const p of products) {
    const size = sizeOf(p.name);
    if (!size) continue;
    const k = [p.brand ?? "", p.segment ?? "", p.name.split(/\s+/).slice(0, 2).join(" ").toLowerCase(), size].join("|");
    groups.set(k, [...(groups.get(k) ?? []), p]);
  }
  // Distinct names only: two EANs with one name are a re-listing, not a sibling.
  return [...groups.values()].map((g) => g.filter((p, i) => g.findIndex((q) => q.name === p.name) === i)).filter((g) => g.length > 1);
}

/** Sibling pair with the largest store-count gap where the narrower one is still seen in at least 5 stores. */
function siblingGap(groups: AccountProduct[][]) {
  const pairs = groups
    .map((g) => {
      const s = [...g].sort((a, b) => b.storesSeen - a.storesSeen);
      return { wide: s[0], narrow: s[s.length - 1] };
    })
    .filter((x) => x.wide.storesSeen > x.narrow.storesSeen);
  const pick = (min: number) => pairs.filter((x) => x.narrow.storesSeen >= min).sort((a, b) => b.wide.storesSeen - b.narrow.storesSeen - (a.wide.storesSeen - a.narrow.storesSeen))[0];
  return pick(5) ?? pick(1) ?? null;
}

                  `Monitors: ${f.monitors.length ? f.monitors.map((m) => `${fmt(m.n)} ${m.origin}${m.status ? ` (${m.status})` : ""}`).join(", ") : "none"}${lastRun ? `, last run ${stamp(lastRun)}` : ""}. Alerts: ${fmt(monitorAlerts)} from monitors, ${fmt(replayAlerts)} replayed.`,
                  `Next expected reactions: ${fmt(ownSoon + compSoon)} promotions (${fmt(ownSoon)} ${company}, ${fmt(compSoon)} competitor) state an end date within 3 days (by ${stamp(new Date(soonBy).toISOString(), false)}).`,
                  liveOn
                    ? `Live monitoring is on with ${plural(f.liveMonitors, "monitor")}.`
                    : `Live monitoring not switched on${!f.hasKey ? " (Context.dev key not set)" : ""}: today this is a ${plural(f.captureDays, "day")} snapshot with history, not a live feed.`,
                ]}
              />
            </tbody>
          </table>
        </div>
      </Card>

      <Card title="What this is not">
        <ul className="grid gap-x-6 gap-y-1.5 text-[13px] text-ink-2 md:grid-cols-2">
          <li>No sales volumes, margins, market share, physical shelf position or in-store stock.</li>
          <li>
            {plural(f.captureDays, "day")} of captures ({stamp(f.firstCapture, false)} to {stamp(f.latestCapture, false)}), so no trends yet.
          </li>
          <li>Helsinki and Espoo stores only.</li>
          <li>A blank store means not captured, not absent from the shelf.</li>
          <li className="md:col-span-2">
            The platform describes the market. It does not recommend coordinating prices with competitors or dictating retailers&apos; resale prices.
          </li>
        </ul>
      </Card>
    </div>
  );
}
